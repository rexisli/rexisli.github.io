const int mod=1e9+7;
class Matrix {
public:	
	int n=0,m=0;
	vector<vector<int>> mat;
	Matrix(int tp_n,int tp_m):n(tp_n),m(tp_m){
		mat.resize(n,vector<int>(m,0));
	}
	void setIdentity(){
		for(int i=0;i<n;i++){
			for(int j=0;j<m;j++){
				mat[i][j]=(i==j?1:0);
			}
		}
	}
	Matrix& operator= (const Matrix& other){
		if (this!=&other) {
			n=other.n;
			m=other.m;
			mat=other.mat;
		}
		return *this;
	}
	Matrix operator* (const Matrix& other) const {
		if(m!=other.n) return Matrix(0,0);
		Matrix ans{n,other.m};
		for(int i=0;i<n;i++){
			for(int j=0;j<other.m;j++){
				int num=0;
				for(int k=0;k<m;k++){
					(num+=mat[i][k]*other.mat[k][j])%=mod;
				}
				ans.mat[i][j]=num;
			}
		}
		return ans;
	}
	friend istream& operator>>(istream& is, Matrix& cur) {
		for(int i=0;i<cur.n;i++){
			for(int j=0;j<cur.m;j++){
				is>>cur.mat[i][j];
				cur.mat[i][j]%=mod;
			}
		}
		return is;
	};
	friend ostream& operator<<(ostream& os, const Matrix& cur) {
		for(int i=0;i<cur.n;i++){
			for(int j=0;j<cur.m;j++){
				if(j) os<<" ";
				os<<cur.mat[i][j];
			}
			os<<endl;
		}
		return os;
	};
};
inline int sum(Matrix x){
	int ans=0;
	for(int i=0;i<x.n;i++){
		for(int j=0;j<x.m;j++){
			(ans+=x.mat[i][j])%=mod;
		}
	}
	return ans;
}
inline Matrix qpow(Matrix base,int b){
	Matrix ans{base.n,base.m};
	ans.setIdentity();
	while(b){
		if(b&1){
			ans=ans*base;
		}
		base=base*base;
		b>>=1;
	}
	return ans;
}
