#include <bits/stdc++.h>
#define mem(x) memset(x,0,sizeof(x))  // 内存初始化宏：将x指向的内存块填充为0
#define int long long                 // 统一将int类型定义为long long，避免整数溢出
#define endl "\n"                     // 输出换行符，比"\n"更高效（避免缓冲区刷新问题）
#define sysp system("pause")          // 程序暂停宏（Windows下有效，用于查看运行结果）
using namespace std;

const int mxn=1e5+10;                // 原始数组的最大长度：1e5+10（预留冗余空间）
const int mxlog2=32;               // ST表第一维的最大长度：log2(1e5)≈17，32足够覆盖所有情况
int n,m;                             // n：原始数组长度；m：查询次数
int a[mxn];                          // 原始数组（存储待查询的初始数据，1-based下标）
int log2_n[mxn];                     // 预处理对数表：log2_n[x]表示floor(log2(x))，即x的整数对数（向下取整）
int ST[mxlog2][mxn];               // ST表核心数组：
// 定义：ST[j][i] 表示「从下标i开始，长度为2^j的区间」内的最大值
// 维度说明：第一维j（对应2^j的区间长度），第二维i（对应区间起始下标）

void init(){
	// ===================== 第一部分：预处理对数表（O(n)时间） =====================
	// 为什么预处理对数表？避免查询时调用log2()浮点数函数（效率低+精度问题），实现O(1)快速查询
	log2_n[2]=1;                     // 初始化：当x=2时，2^1=2，故floor(log2(2))=1
	// 循环递推预处理x>=3的对数表
	for(int i=3;i<=n;i++){
		// 核心转移式：log2_n[i] = log2_n[i>>1] + 1
		// 数学证明：
		// 1.  i>>1 等价于 整数除法 i/2（向下取整）
		// 2.  对于任意整数i>=3，log2(i) = log2(2 * (i/2)) = 1 + log2(i/2)
		// 3.  对等式两边向下取整：floor(log2(i)) = floor(1 + log2(i/2)) = 1 + floor(log2(i/2))
		// 4.  即 log2_n[i] = log2_n[i/2] + 1 = log2_n[i>>1] + 1
		log2_n[i]=log2_n[i>>1]+1;
	}
	
	// ===================== 第二部分：构建ST表（O(n log n)时间） =====================
	// 步骤1：初始化ST表的第0层（j=0，对应区间长度2^0=1）
	for(int i=1;i<=n;i++){
		// 当j=0时，区间长度为1，区间[i, i+2^0-1] = [i,i]，最大值就是原始数组a[i]本身
		ST[0][i]=a[i];
	}
	
	// 步骤2：递推构建ST表的第1~log2_n[n]层（按区间长度从小到大递推）
	// j的含义：当前构建的ST表层号，对应区间长度为len_j = 2^j
	// j的取值范围：1 <= j <= log2_n[n]，因为最大的有效区间长度不超过n（2^log2_n[n] <= n < 2^(log2_n[n]+1)）
	for(int j=1;j<=log2_n[n];j++){// ST表第一维（层号j）：从短区间向长区间递推
		// i的含义：当前区间的起始下标（1-based）
		// 循环条件：i + (1<<j) - 1 <= n
		// 含义解释：(1<<j) = 2^j（当前区间长度），i + 2^j - 1 是当前区间的右端点
		// 该条件保证：区间[i, i+2^j-1] 完全落在原始数组[1,n]范围内（右端点不越界），确保ST[j][i]有效
		for(int i=1;i+(1<<j)-1<=n;i++){// ST表第二维（起始下标i）：枚举所有合法起始位置
			// 核心递推式：ST[j][i] = max(ST[j-1][i], ST[j-1][i + (1<<(j-1))])
			// 先明确关键变量：
			//   len_j = 2^j（当前j层区间长度）
			//   len_j_1 = 2^(j-1) = 1<<(j-1)（j-1层区间长度，是j层的一半）
			// 数学证明（递推合法性）：
			// 1.  ST[j][i] 对应区间：[L1, R1] = [i, i + len_j - 1] = [i, i + 2^j - 1]
			// 2.  ST[j-1][i] 对应区间：[L2, R2] = [i, i + len_j_1 - 1] = [i, i + 2^(j-1) - 1]
			// 3.  ST[j-1][i + len_j_1] 对应区间：[L3, R3] = [i+2^(j-1), i+2^(j-1)+2^(j-1)-1] = [i+2^(j-1), i+2^j -1]
			// 4.  显然 [L2, R2] ∪ [L3, R3] = [L1, R1]，且两个子区间无遗漏、无冲突（刚好拼接成父区间）
			// 5.  因此，父区间的最大值 = 两个子区间最大值的较大者，即上述递推式
			ST[j][i]=max(ST[j-1][i], ST[j-1][i+(1<<(j-1))]);
		}
	}
}

// 区间最大值查询函数（O(1)时间）
// 参数：l -> 查询区间左端点，r -> 查询区间右端点（均为1-based下标）
int query(int l,int r){
	// 步骤1：计算查询区间的长度
	int len = r - l + 1;
	// 步骤2：获取len对应的整数对数s = floor(log2(len))，即最大的s满足 2^s <= len
	int s=log2_n[len];
	// 步骤3：查询两个重叠区间的最大值，作为最终结果
	// 核心式子：max(ST[s][l], ST[s][r - (1<<s) + 1])
	// 数学证明（结果正确性）：
	// 1.  第一个区间：ST[s][l] -> [l, l + 2^s - 1]
	// 2.  第二个区间：ST[s][r - 2^s + 1] -> [r - 2^s + 1, (r - 2^s + 1) + 2^s - 1] = [r - 2^s + 1, r]
	// 3.  证明两个区间的并集覆盖查询区间[l, r]：
	//    因为 2^s <= len = r-l+1，所以 r - 2^s + 1 <= l + (r-l+1) - 2^s + 1 - l = r+1 -2^s <= r+1 -1 = r
	//    同时，l + 2^s - 1 >= l + (r-l+1) -1 = r（因为2^s <= len，此处为重叠关键）
	//    因此 [l, l+2^s-1] 和 [r-2^s+1, r] 必然重叠，且并集恰好是[l, r]
	// 4.  区间最大值具有单调性：覆盖查询区间的两个区间的最大值的较大者，就是查询区间的最大值
	return max(ST[s][l], ST[s][r-(1<<s)+1]);
}

signed main(){
	// 输入输出优化：关闭C++标准流与C标准流的同步，解绑cin和cout
	// 作用：大幅提高cin/cout的读写速度，避免大数据量下超时
	ios::sync_with_stdio(false);
	cin.tie(0),cout.tie(0);
	
	cin>>n>>m; // 输入原始数组长度n，查询次数m
	// 输入原始数组a（1-based下标，与ST表定义一致）
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	
	init(); // 初始化对数表 + 构建ST表
	
	// 处理m次查询
	for(int i=1;i<=m;i++){
		int l,r;
		cin>>l>>r; // 输入每次查询的左右端点[l,r]
		cout<<query(l,r)<<endl; // 输出查询结果并换行
	}
	return 0;
}
