class BigInt {
private:
	vector<int> digits;  // 倒序存储数字，每位0-9
	bool is_negative;    // 符号标志
	static const size_t KARATSUBA_THRESHOLD = 32;  // Karatsuba算法阈值（小数字用竖式乘法）
	// 内部工具函数
	void remove_leading_zeros() {
		while (digits.size() > 1 && digits.back() == 0) digits.pop_back();
		if (is_zero()) is_negative = false;
	};
	void normalize() {
		remove_leading_zeros();
	};
	int compare_abs(const BigInt& other) const {
		if (digits.size() != other.digits.size()) return digits.size() < other.digits.size() ? -1 : 1;
		for (int i = digits.size() - 1; i >= 0; i--) if (digits[i] != other.digits[i]) return digits[i] < other.digits[i] ? -1 : 1;
		return 0;
	};
	BigInt shift_left(size_t m) const { // 左移m位（乘以10^m）
		if (is_zero()) return BigInt(0);
		BigInt res = *this;
		res.digits.insert(res.digits.end(), m, 0);  // 倒序存储，末尾加0即高位扩展
		res.normalize();
		return res;
	};
	BigInt get_high(size_t k) const { // 获取高位k位
		BigInt res;
		res.is_negative = false, res.digits.clear();
		if (k >= digits.size()) {
			res.digits.push_back(0);
			return res;
		}
		for (size_t i = k; i < digits.size(); ++i) res.digits.push_back(digits[i]); // 从第k位开始取到末尾（高位部分）
		res.normalize();
		return res;
	};
	BigInt get_low(size_t k) const { // 获取低位k位
		BigInt res;
		res.is_negative = false, res.digits.clear();
		size_t len = min(k, digits.size());
		for (size_t i = 0; i < len; ++i) res.digits.push_back(digits[i]); // 取前k位（低位部分）
		res.normalize();
		return res;
	};
public:
	// 构造函数
	BigInt() : is_negative(false) {
		digits.push_back(0);
	}
	BigInt(long long num) {
		if (num < 0) is_negative = true, num = -num;
		else is_negative = false;
		if (num == 0) digits.push_back(0);
		else while (num > 0) digits.push_back(num % 10), num /= 10;
	}
	BigInt(const string& str) {
		if (str.empty()) {
			digits.push_back(0), is_negative = false;
			return;
		}
		size_t start = 0;
		if (str[0] == '-') is_negative = true, start = 1;
		else if (str[0] == '+')	is_negative = false, start = 1;
		else is_negative = false;
		while (start < str.size() && str[start] == '0') start++; // 跳过前导零
		if (start == str.size()) {
			digits.push_back(0), is_negative = false;
			return;
		}
		// 预分配空间提高效率
		digits.reserve(str.size() - start);
		for (int i = str.size() - 1; i >= static_cast<int>(start); i--) {
			if (!isdigit(str[i])) throw invalid_argument("Invalid character in number string");
			digits.push_back(str[i] - '0');
		}
		remove_leading_zeros();
	};
	BigInt(const BigInt& other) = default;
	BigInt& operator=(const BigInt& other) = default;
	// 基本信息
	bool is_zero() const {
		return digits.size() == 1 && digits[0] == 0;
	}
	BigInt abs() const {
		BigInt result = *this;
		result.is_negative = false;
		return result;
	};
	// 比较运算符
	bool operator==(const BigInt& other) const {
		if (is_zero() && other.is_zero()) return true;
		if (is_negative != other.is_negative) return false;
		return compare_abs(other) == 0;
	};
	bool operator!=(const BigInt& other) const {
		return !(*this == other);
	}
	bool operator<(const BigInt& other) const {
		if (is_negative != other.is_negative) return is_negative;
		int cmp = compare_abs(other);
		return is_negative ? (cmp > 0) : (cmp < 0);
	};
	bool operator<=(const BigInt& other) const {
		return *this < other || *this == other;
	}
	bool operator>(const BigInt& other) const {
		return !(*this <= other);
	}
	bool operator>=(const BigInt& other) const {
		return !(*this < other);
	}
	// 算术运算符
	BigInt operator-() const {
		BigInt result = *this;
		if (!result.is_zero()) result.is_negative = !result.is_negative;
		return result;
	};
	BigInt operator+(const BigInt& other) const {
		if (is_negative != other.is_negative) return is_negative ? (other - (-*this)) : (*this - (-other));// 处理符号不同的情况：转换为减法
		BigInt result;
		result.is_negative = is_negative, result.digits.clear();
		const size_t max_size = max(digits.size(), other.digits.size());
		result.digits.reserve(max_size + 1);
		int carry = 0;
		for (size_t i = 0; i < max_size || carry; i++) {
			int sum = carry;
			if (i < digits.size()) sum += digits[i];
			if (i < other.digits.size()) sum += other.digits[i];
			result.digits.push_back(sum % 10), carry = sum / 10;
		}
		result.normalize();
		return result;
	};
	BigInt operator-(const BigInt& other) const {
		if (is_negative != other.is_negative) return *this + (-other); // 处理符号不同的情况：转换为加法
		if (compare_abs(other) < 0) { // 同号相减，确保被减数的绝对值大于减数
			BigInt result = other - *this;
			result.is_negative = !is_negative;
			return result;
		}
		BigInt result;
		result.is_negative = is_negative, result.digits.clear(), result.digits.reserve(digits.size());
		int borrow = 0;
		for (size_t i = 0; i < digits.size(); i++) {
			int diff = digits[i] - borrow;
			if (i < other.digits.size()) diff -= other.digits[i];
			if (diff < 0) diff += 10, borrow = 1;
			else borrow = 0;
			result.digits.push_back(diff);
		}
		result.normalize();
		return result;
	};
	BigInt operator*(const BigInt& other) const {
		// 优化：Karatsuba乘法（O(n^log3)≈O(n^1.58)）
		if (is_zero() || other.is_zero()) return BigInt(0);
		// 小数字用原竖式乘法（避免递归开销）
		if (digits.size() < KARATSUBA_THRESHOLD || other.digits.size() < KARATSUBA_THRESHOLD) {
			BigInt result;
			result.is_negative = (is_negative != other.is_negative), result.digits.resize(digits.size() + other.digits.size(), 0);
			for (size_t i = 0; i < digits.size(); i++) {
				int carry = 0;
				for (size_t j = 0; j < other.digits.size() || carry; j++) {
					long long product = result.digits[i + j] + static_cast<long long>(digits[i]) * (j < other.digits.size() ? other.digits[j] : 0) + carry;
					result.digits[i + j] = product % 10;
					carry = product / 10;
				}
			}
			result.normalize();
			return result;
		}
		// Karatsuba核心逻辑：a = a1*10^m + a0, b = b1*10^m + b0
		size_t n = max(digits.size(), other.digits.size());
		size_t m = (n + 1) / 2;  // 分割点（高位m位，低位n-m位）
		BigInt a0 = get_low(m);    // a的低位
		BigInt a1 = get_high(m);   // a的高位
		BigInt b0 = other.get_low(m);  // b的低位
		BigInt b1 = other.get_high(m); // b的高位
		// 3次乘法替代4次：z0=a0*b0, z1=(a0+a1)(b0+b1), z2=a1*b1
		BigInt z0 = a0 * b0;
		BigInt z1 = (a0 + a1) * (b0 + b1);
		BigInt z2 = a1 * b1;
		// 结果 = z2*10^(2m) + (z1-z2-z0)*10^m + z0
		BigInt result = z2.shift_left(2 * m) + (z1 - z2 - z0).shift_left(m) + z0;
		result.is_negative = (is_negative != other.is_negative), result.normalize();
		return result;
	};
	BigInt operator/(const BigInt& other) const {
		if (other.is_zero()) throw invalid_argument("Division by zero");
		if (is_zero()) return BigInt(0);
		BigInt dividend = abs();
		BigInt divisor = other.abs();
		if (dividend < divisor) return BigInt(0);
		BigInt quotient;
		quotient.is_negative = (is_negative != other.is_negative), quotient.digits.clear();
		BigInt current;
		for (int i = dividend.digits.size() - 1; i >= 0; i--) {
			current.digits.insert(current.digits.begin(), dividend.digits[i]), current.normalize();
			if (current < divisor) {
				quotient.digits.insert(quotient.digits.begin(), 0);
				continue;
			}
			int left = 0, right = 9, digit = 0;
			while (left <= right) {
				int mid = (left + right) / 2;
				BigInt temp_product = divisor * BigInt(mid);
				if (temp_product <= current) digit = mid, left = mid + 1;
				else right = mid - 1;
			}
			quotient.digits.insert(quotient.digits.begin(), digit);
			current = current - (divisor * BigInt(digit));
		}
		quotient.normalize();
		return quotient;
	};
	BigInt operator%(const BigInt& other) const {
		if (other.is_zero()) throw invalid_argument("Modulo by zero");
		BigInt quotient = *this / other;
		BigInt remainder = *this - quotient * other;
		return remainder;
	};
	// 输入输出
	friend ostream& operator<<(ostream& os, const BigInt& num) {
		if (num.is_negative) os << '-';
		for (int i = num.digits.size() - 1; i >= 0; i--) os << num.digits[i];
		return os;
	};
	friend istream& operator>>(istream& is, BigInt& num) {
		string str;
		is >> str, num = BigInt(str);
		return is;
	};
};
